# Approach

we store the positions of `*` in each row and column using two maps. then we find the row and column that contain exactly one `*` and output their 1-based indices.

time complexity - `O(n * m)`
space complexity - `O(n * m)` auxiliary space

# Code

```cpp
#include<bits/stdc++.h>
using namespace std;

int main(){
    int n = 0, m = 0;
    cin >> n >> m;

    vector<vector<char>> arr(n, vector<char>(m));
    unordered_map<int, int> mpprow;
    unordered_map<int, int> mppcol;

    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            cin >> arr[i][j];
            if(arr[i][j] == '*'){
                mpprow[i]++;
                mppcol[j]++;
            }
        }
    }

    for(auto i : mpprow){
        if(i.second == 1) cout << i.first + 1;
    }

    cout << ' ';

    for(auto i : mppcol){
        if(i.second == 1) cout << i.first + 1;
    }
}
