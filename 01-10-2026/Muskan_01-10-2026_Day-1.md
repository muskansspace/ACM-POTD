Approach
traversing the array to find the first row and the last row with * as well as the first column and the last column with *
for minirow we are checking only once that is our first encounter with * but for minicol we need to keep checking in every row as * could appear early than prev rows. 
the entire approach takes O(n * m) for determining the boundary of the rectangle and another O(n * m) for printing the output - O(n * m)
space complexity - O(n * m) for storing the input

Code
#include<bits/stdc++.h>
using namespace std;
int main(){
    int n=0; int m=0; 
    cin>>n; cin>>m;
    
    vector<vector<char>> arr(n, vector<char>(m));
    int minirow = -1; int maxirow = 0; int minicol = INT_MAX; int maxicol = 0;
    for(int i=0; i<n; i++){
        for(int j=0;j<m;j++){
            cin>>arr[i][j];
            if(arr[i][j] == '*'){
                if(minirow == -1) minirow = i;
                maxirow = max(maxirow, i);
                minicol = min(minicol, j);
                maxicol = max(maxicol, j); 
            }
        }
    }
    
    for(int i = minirow; i <= maxirow; i++){
        for(int j = minicol; j <= maxicol; j++){
            cout << arr[i][j];
        }
        cout << endl;
    }
}
